#!/usr/bin/env python3
import os
import glob
import time
import threading
import sys

# LCD display library for 16x2 character LCD via I2C
from RPLCD.i2c import CharLCD

# SPI communication for MCP3008 ADC
import spidev

# Camera, YOLO object detection, and computer vision libraries
import pathlib
import platform
if platform.system() == "Linux":
    pathlib.WindowsPath = pathlib.PosixPath
import cv2
import torch
import numpy as np

# Raspberry Pi Camera v2 interface
from picamera2 import Picamera2

# YOLOv5 model and utilities for food detection
from models.common import DetectMultiBackend
from utils.torch_utils import select_device
from utils.general import non_max_suppression, scale_boxes
from utils.augmentations import letterbox

# Configuration: LCD I2C address
LCD_I2C_ADDR = 0x27
LCD_LOCK = threading.Lock()  # Thread-safe LCD access
LCD = CharLCD('PCF8574', LCD_I2C_ADDR)

# DS18B20 temperature sensor initialization via 1-Wire protocol
os.system('modprobe w1-gpio')
os.system('modprobe w1-therm')
BASE_DIR = '/sys/bus/w1/devices/'

# MCP3008 ADC initialization via SPI for pH and gas sensors
spi = spidev.SpiDev()
spi.open(0, 0)
spi.max_speed_hz = 1350000
PH_CHANNEL = 0  # pH sensor connected to channel 0
MQ_CHANNEL = 1  # MQ gas sensor connected to channel 1

# YOLOv5 model initialization for food object detection
device = select_device('cpu')
model = DetectMultiBackend('best.pt', device=device)
model.eval()

# Picamera2 and thread control variables
picam2 = None
STOP_THREADS = False

def find_ds18b20_device_folder():
    """Locates DS18B20 temperature sensor device folder in 1-Wire bus"""
    try:
        folders = glob.glob(BASE_DIR + '28-*')
        if not folders:
            return None
        return folders[0]
    except Exception:
        return None

def read_ds18b20_raw(device_file):
    """Reads raw data from DS18B20 sensor file"""
    with open(device_file, 'r') as f:
        return f.readlines()

def read_ds18b20_temp():
    """Reads temperature from DS18B20 sensor and returns value in Celsius"""
    device_folder = find_ds18b20_device_folder()
    if not device_folder:
        return None
    device_file = device_folder + '/w1_slave'
    try:
        lines = read_ds18b20_raw(device_file)
    except Exception:
        return None
    # Wait for valid sensor reading (YES flag)
    attempts = 0
    while lines[0].strip()[-3:] != 'YES' and attempts < 5:
        time.sleep(0.2)
        try:
            lines = read_ds18b20_raw(device_file)
        except Exception:
            return None
        attempts += 1
    if lines[0].strip()[-3:] != 'YES':
        return None
    # Extract temperature value from sensor data
    equals_pos = lines[1].find('t=')
    if equals_pos != -1:
        temp_string = lines[1][equals_pos + 2:]
        try:
            temp_c = float(temp_string) / 1000.0
            return temp_c
        except Exception:
            return None
    return None
def read_adc(channel):
    """Reads analog value from MCP3008 ADC channel (0-7) via SPI"""
    if channel < 0 or channel > 7:
        return -1
    adc = spi.xfer2([1, (8 + channel) << 4, 0])
    data = ((adc[1] & 3) << 8) + adc[2]
    return data
def adc_to_voltage(adc_value, vref=3.3):
    """Converts 10-bit ADC value to voltage based on reference voltage"""
    return (adc_value * vref) / 1023.0
def get_ph_status_from_voltage(voltage_ph):
    """Determines food freshness based on pH sensor voltage"""
    if voltage_ph >= 0.5 and voltage_ph <= 3.0:
        ph_status = "Spoiled"
    elif voltage_ph >= 3.0 and voltage_ph <= 3.5:
        ph_status = "SPOILED"
    else:
        ph_status = "Fresh"
    return ph_status

def get_mq_status_from_voltage(voltage_mq):
    """Determines food freshness based on MQ gas sensor voltage"""
    if 1.0 <= voltage_mq <= 1.5:
        mq_status = "Fresh"
    elif 2.50 <= voltage_mq <= 3.30:
        mq_status = "Spoiled"
    else:
        mq_status = "Sensor Is In Air"
    return mq_status

def lcd_write_lines(line1, line2):
    """Thread-safe function to write two lines to 16x2 LCD display"""
    with LCD_LOCK:
        try:
            LCD.clear()
            LCD.write_string(line1[:16])
            LCD.cursor_pos = (1, 0)
            LCD.write_string(line2[:16])
        except Exception as e:
            print(f"[LCD ERROR] {e}")

def sensor_lcd_worker():
    """Background thread that continuously reads sensors and updates LCD display"""
    global STOP_THREADS
    while not STOP_THREADS:
        # Display DS18B20 temperature reading
        temp = read_ds18b20_temp()
        if temp is not None:
            line1 = "FOOD STATUS"
            line2 = f"Temp: {temp:.1f}C"
        else:
            line1 = "FOOD STATUS"
            line2 = "DS18B20 ERROR"
        lcd_write_lines(line1, line2)
        print(f"[DS18B20] {line2}")
        for _ in range(20):
            if STOP_THREADS:
                break
            time.sleep(0.1)
        if STOP_THREADS:
            break
        # Display pH sensor reading and status
        adc_ph = read_adc(PH_CHANNEL)
        voltage_ph = adc_to_voltage(adc_ph)
        ph_status = get_ph_status_from_voltage(voltage_ph)
        lcd_write_lines("FOOD STATUS", "PH_Sensor: " + ph_status)
        print(f"[pH] ADC: {adc_ph} | Voltage: {voltage_ph:.2f} V | Status: {ph_status}")
        for _ in range(20):
            if STOP_THREADS:
                break
            time.sleep(0.1)
        if STOP_THREADS:
            break
        # Display MQ gas sensor reading and status
        adc_mq = read_adc(MQ_CHANNEL)
        voltage_mq = adc_to_voltage(adc_mq)
        mq_status = get_mq_status_from_voltage(voltage_mq)
        lcd_write_lines("FOOD STATUS", "MQ_S : " + mq_status)
        print(f"[MQ] ADC: {adc_mq} | Voltage: {voltage_mq:.2f} V | Status: {mq_status}")
        for _ in range(20):
            if STOP_THREADS:
                break
            time.sleep(0.1)
    # Cleanup LCD when thread stops
    with LCD_LOCK:
        try:
            LCD.clear()
            LCD.write_string("Sensor thread stopped")
            time.sleep(1)
            LCD.clear()
        except Exception:
            pass
def camera_yolo_loop():
    """Main loop for camera capture and YOLOv5 object detection"""
    global picam2, STOP_THREADS
    # Initialize Picamera2 with 640x480 resolution
    try:
        picam2 = Picamera2()
        camera_config = picam2.create_preview_configuration(main={"size": (640, 480)})
        picam2.configure(camera_config)
        picam2.start()
        time.sleep(1)
        print("Picamera2 started.")
    except Exception as e:
        print(f"FATAL ERROR: Picamera2 failed to start: {e}")
        STOP_THREADS = True
        return
    try:
        while not STOP_THREADS:
            # Capture frame from camera
            frame = picam2.capture_array()
            frame = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
            # Preprocess image for YOLO model input
            img = letterbox(frame, new_shape=(640, 640))[0]
            img = img[:, :, ::-1].transpose(2, 0, 1)
            img = np.ascontiguousarray(img)
            img = torch.from_numpy(img).float()
            img /= 255.0
            if img.ndimension() == 3:
                img = img.unsqueeze(0).to(device)
            else:
                img = img.to(device)
            # Run YOLO inference
            with torch.no_grad():
                pred = model(img, augment=False, visualize=False)
            pred = non_max_suppression(pred, 0.25, 0.45, None, False, max_det=1000)
            # Draw bounding boxes and labels on detected objects
            for det in pred:
                if len(det):
                    det[:, :4] = scale_boxes(img.shape[2:], det[:, :4], frame.shape).round()
                    for *xyxy, conf, cls in det:
                        x1, y1, x2, y2 = map(int, xyxy)
                        label = f"{model.names[int(cls)]} {conf:.2f}"
                        cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)
                        cv2.putText(frame, label, (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)
            # Display annotated frame
            cv2.imshow("YOLOv5 Real-Time Detection", frame)
            if cv2.waitKey(1) & 0xFF == ord('q'):
                STOP_THREADS = True
                break
    except Exception as e:
        print(f"Camera loop error: {e}")
        STOP_THREADS = True
    finally:
        # Cleanup camera resources
        try:
            if picam2 is not None:
                picam2.stop()
        except Exception:
            pass
        cv2.destroyAllWindows()
if __name__ == "__main__":
    try:
        # Display startup message on LCD
        with LCD_LOCK:
            try:
                LCD.clear()
                LCD.write_string("pH & Food Monitor")
                time.sleep(1.5)
                LCD.clear()
            except Exception:
                pass
        # Start sensor monitoring thread
        t_sensor = threading.Thread(target=sensor_lcd_worker, daemon=True)
        t_sensor.start()
        # Start camera and YOLO detection in main thread
        camera_yolo_loop()
    except KeyboardInterrupt:
        STOP_THREADS = True
        print("\nProgram stopped by user (KeyboardInterrupt).")
    except Exception as e:
        STOP_THREADS = True
        print(f"\nUnexpected error: {e}")
    finally:
        # Graceful shutdown: stop all threads and cleanup resources
        STOP_THREADS = True
        time.sleep(0.5)
        try:
            spi.close()
        except Exception:
            pass
        with LCD_LOCK:
            try:
                LCD.clear()
                LCD.write_string("System Stopped")
                time.sleep(1)
                LCD.clear()
            except Exception:
                pass
        try:
            if picam2 is not None:
                picam2.stop()
        except Exception:
            pass
        try:
            cv2.destroyAllWindows()
        except Exception:
            pass
        print("Clean exit.")
